"""
locator_utils.py - Locator utility functions for Selenium Recorder

This module provides utility functions for generating and managing element locators.
"""

import re
from typing import Dict, List, Any, Optional, Tuple
import logging

logger = logging.getLogger("LocatorUtils")

class LocatorUtils:
    """
    Utility class for generating and managing element locators.
    """
    
    @staticmethod
    def generate_locators(element_info: Dict[str, Any]) -> Dict[str, Any]:
        """
        Generate locators from element information.
        
        Args:
            element_info: Dictionary containing element information
            
        Returns:
            Dictionary of locators
        """
        locators = {}
        
        # ID locator (most reliable)
        if element_info.get("id"):
            locators["id"] = element_info["id"]
            
        # Name locator
        if element_info.get("name"):
            locators["name"] = element_info["name"]
            
        # Class name locator
        if element_info.get("class_name"):
            locators["class_name"] = element_info["class_name"]
            
        # Tag name locator
        if element_info.get("tag_name"):
            locators["tag_name"] = element_info["tag_name"]
            
        # Link text locator (for anchor tags)
        if element_info.get("tag_name") == "a" and element_info.get("text"):
            locators["link_text"] = element_info["text"]
            
        # XPath locators
        locators["xpath"] = LocatorUtils.generate_xpath_locators(element_info)
        
        # CSS selector
        locators["css"] = LocatorUtils.generate_css_selector(element_info)
        
        return locators
        
    @staticmethod
    def generate_xpath_locators(element_info: Dict[str, Any]) -> Dict[str, str]:
        """
        Generate different XPath locators for an element.
        
        Args:
            element_info: Dictionary containing element information
            
        Returns:
            Dictionary of XPath locators
        """
        xpath_locators = {}
        
        # ID-based XPath (most reliable)
        if element_info.get("id"):
            xpath_locators["id_based"] = f"//*[@id='{element_info['id']}']"
            
        # Name-based XPath
        if element_info.get("name"):
            tag_name = element_info.get("tag_name", "*")
            xpath_locators["name_based"] = f"//{tag_name}[@name='{element_info['name']}']"
            
        # Class-based XPath
        if element_info.get("class_name"):
            tag_name = element_info.get("tag_name", "*")
            xpath_locators["class_based"] = f"//{tag_name}[@class='{element_info['class_name']}']"
            
        # Text-based XPath
        if element_info.get("text"):
            tag_name = element_info.get("tag_name", "*")
            xpath_locators["text_based"] = f"//{tag_name}[text()='{element_info['text']}']"
            xpath_locators["contains_text"] = f"//{tag_name}[contains(text(),'{element_info['text']}')]"
            
        # Attribute-based XPath
        if element_info.get("attributes"):
            tag_name = element_info.get("tag_name", "*")
            attrs = element_info["attributes"]
            
            # Try to find unique attributes
            for attr_name in ["data-testid", "data-id", "data-automation", "aria-label"]:
                if attr_name in attrs:
                    xpath_locators[f"{attr_name}_based"] = f"//{tag_name}[@{attr_name}='{attrs[attr_name]}']"
                    
        # Use full XPath as fallback
        if element_info.get("xpath"):
            xpath_locators["full"] = element_info["xpath"]
            
        return xpath_locators
        
    @staticmethod
    def generate_css_selector(element_info: Dict[str, Any]) -> str:
        """
        Generate CSS selector for an element.
        
        Args:
            element_info: Dictionary containing element information
            
        Returns:
            CSS selector
        """
        # ID selector (most reliable)
        if element_info.get("id"):
            return f"#{element_info['id']}"
            
        # Name selector
        if element_info.get("name"):
            tag_name = element_info.get("tag_name", "*")
            return f"{tag_name}[name='{element_info['name']}']"
            
        # Class selector
        if element_info.get("class_name"):
            tag_name = element_info.get("tag_name", "*")
            # Handle multiple classes
            classes = element_info["class_name"].split()
            if classes:
                return f"{tag_name}.{'.'.join(classes)}"
                
        # Attribute selectors
        if element_info.get("attributes"):
            tag_name = element_info.get("tag_name", "*")
            attrs = element_info["attributes"]
            
            # Try to find unique attributes
            for attr_name in ["data-testid", "data-id", "data-automation", "aria-label"]:
                if attr_name in attrs:
                    return f"{tag_name}[{attr_name}='{attrs[attr_name]}']"
                    
        # Fallback to tag name
        if element_info.get("tag_name"):
            return element_info["tag_name"]
            
        return "*"  # Universal selector as last resort
        
    @staticmethod
    def rank_locators(locators: Dict[str, Any]) -> List[Tuple[str, str, int]]:
        """
        Rank locators by reliability.
        
        Args:
            locators: Dictionary of locators
            
        Returns:
            List of tuples (locator_type, locator_value, score)
        """
        ranked_locators = []
        
        # ID locator (most reliable)
        if locators.get("id"):
            ranked_locators.append(("id", locators["id"], 100))
            
        # XPath ID-based
        if locators.get("xpath", {}).get("id_based"):
            ranked_locators.append(("xpath", locators["xpath"]["id_based"], 95))
            
        # CSS ID selector
        if locators.get("css") and locators["css"].startswith("#"):
            ranked_locators.append(("css", locators["css"], 90))
            
        # Name locator
        if locators.get("name"):
            ranked_locators.append(("name", locators["name"], 85))
            
        # XPath name-based
        if locators.get("xpath", {}).get("name_based"):
            ranked_locators.append(("xpath", locators["xpath"]["name_based"], 80))
            
        # Data attribute XPath
        for attr in ["data-testid_based", "data-id_based", "data-automation_based"]:
            if locators.get("xpath", {}).get(attr):
                ranked_locators.append(("xpath", locators["xpath"][attr], 75))
                
        # Link text
        if locators.get("link_text"):
            ranked_locators.append(("link_text", locators["link_text"], 70))
            
        # XPath text-based
        if locators.get("xpath", {}).get("text_based"):
            ranked_locators.append(("xpath", locators["xpath"]["text_based"], 65))
            
        # Class-based locators
        if locators.get("class_name"):
            ranked_locators.append(("class_name", locators["class_name"], 60))
            
        if locators.get("xpath", {}).get("class_based"):
            ranked_locators.append(("xpath", locators["xpath"]["class_based"], 55))
            
        # XPath contains text
        if locators.get("xpath", {}).get("contains_text"):
            ranked_locators.append(("xpath", locators["xpath"]["contains_text"], 50))
            
        # Tag name (least reliable)
        if locators.get("tag_name"):
            ranked_locators.append(("tag_name", locators["tag_name"], 30))
            
        # Full XPath (fallback)
        if locators.get("xpath", {}).get("full"):
            ranked_locators.append(("xpath", locators["xpath"]["full"], 20))
            
        # Sort by score (descending)
        return sorted(ranked_locators, key=lambda x: x[2], reverse=True)
        
    @staticmethod
    def get_best_locator(locators: Dict[str, Any]) -> Tuple[str, str]:
        """
        Get the best locator from a set of locators.
        
        Args:
            locators: Dictionary of locators
            
        Returns:
            Tuple of (locator_type, locator_value)
        """
        ranked = LocatorUtils.rank_locators(locators)
        
        if ranked:
            return ranked[0][0], ranked[0][1]
            
        # Fallback to full XPath if available
        if locators.get("xpath", {}).get("full"):
            return "xpath", locators["xpath"]["full"]
            
        # Last resort
        return "css", "*"
        
    @staticmethod
    def clean_locator_value(locator_type: str, locator_value: str) -> str:
        """
        Clean and normalize locator value.
        
        Args:
            locator_type: Type of locator
            locator_value: Locator value
            
        Returns:
            Cleaned locator value
        """
        if not locator_value:
            return ""
            
        # Clean XPath
        if locator_type == "xpath":
            # Remove redundant whitespace
            locator_value = re.sub(r'\s+', ' ', locator_value).strip()
            
        # Clean CSS selector
        elif locator_type == "css":
            # Remove redundant whitespace
            locator_value = re.sub(r'\s+', ' ', locator_value).strip()
            
        # Clean text-based locators
        elif locator_type in ["link_text", "partial_link_text"]:
            # Trim and limit length
            locator_value = locator_value.strip()
            if len(locator_value) > 100:
                locator_value = locator_value[:97] + "..."
                
        return locator_value
